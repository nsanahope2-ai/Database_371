CREATE OR REPLACE FUNCTION fn_free_beds(p_room_id INT)
RETURNS INT
LANGUAGE plpgsql
AS $$
DECLARE
    v_capacity     INT;
    v_active_count INT;
BEGIN
    IF p_room_id IS NULL THEN
        RAISE EXCEPTION 'Room ID cannot be NULL.';
    END IF;

    SELECT r.capacity
      INTO v_capacity
      FROM rooms r
     WHERE r.room_id = p_room_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Unknown room_id: % does not exist.', p_room_id;
    END IF;

    SELECT COUNT(*)::INT
      INTO v_active_count
      FROM allocations a
     WHERE a.room_id = p_room_id
       AND a.status = 'ACTIVE';

    RETURN v_capacity - v_active_count;
END;
$$;

CREATE OR REPLACE PROCEDURE pr_allocate_room(
    p_student_id   INT,
    p_room_id      INT,
    p_allocated_on DATE
)
LANGUAGE plpgsql
AS $$
BEGIN
    IF p_student_id IS NULL THEN
        RAISE EXCEPTION 'p_student_id cannot be NULL.';
    END IF;

    IF p_room_id IS NULL THEN
        RAISE EXCEPTION 'p_room_id cannot be NULL.';
    END IF;

    IF p_allocated_on IS NULL THEN
        RAISE EXCEPTION 'p_allocated_on date cannot be NULL.';
    END IF;

    -- Validate student exists and lock student row
    PERFORM 1
       FROM students
      WHERE student_id = p_student_id
        FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Unknown student_id: % does not exist.', p_student_id;
    END IF;

    -- Validate room exists and lock room row before checking bed availability
    PERFORM 1
       FROM rooms
      WHERE room_id = p_room_id
        FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Unknown room_id: % does not exist.', p_room_id;
    END IF;

    -- Ensure the student has no ACTIVE allocation
    IF EXISTS (
        SELECT 1
          FROM allocations
         WHERE student_id = p_student_id
           AND status = 'ACTIVE'
    ) THEN
        RAISE EXCEPTION 'Student % already has an ACTIVE hostel allocation.', p_student_id;
    END IF;

    -- Check room bed availability
    IF fn_free_beds(p_room_id) <= 0 THEN
        RAISE EXCEPTION 'Room % is full; no free beds available.', p_room_id;
    END IF;

    -- Create one ACTIVE allocation with released_on NULL
    INSERT INTO allocations (student_id, room_id, allocated_on, released_on, status)
    VALUES (p_student_id, p_room_id, p_allocated_on, NULL, 'ACTIVE');
END;
$$;

CREATE OR REPLACE FUNCTION fn_trg_enforce_allocation_rules()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
    v_capacity       INT;
    v_active_in_room INT;
BEGIN
    IF NEW.status = 'ACTIVE' THEN
        -- Lock target room row to prevent concurrent over-allocation
        SELECT r.capacity
          INTO v_capacity
          FROM rooms r
         WHERE r.room_id = NEW.room_id
           FOR UPDATE;

        IF NOT FOUND THEN
            RAISE EXCEPTION 'Room ID % does not exist.', NEW.room_id;
        END IF;

        -- Enforce at most one ACTIVE allocation per student (exclude self on UPDATE)
        IF EXISTS (
            SELECT 1
              FROM allocations a
             WHERE a.student_id = NEW.student_id
               AND a.status = 'ACTIVE'
               AND (TG_OP = 'INSERT' OR a.allocation_id <> NEW.allocation_id)
        ) THEN
            RAISE EXCEPTION 'Trigger prevented SQL change: Student % already has an ACTIVE allocation.', NEW.student_id;
        END IF;

        -- Enforce room capacity (exclude self on UPDATE)
        SELECT COUNT(*)::INT
          INTO v_active_in_room
          FROM allocations a
         WHERE a.room_id = NEW.room_id
           AND a.status = 'ACTIVE'
           AND (TG_OP = 'INSERT' OR a.allocation_id <> NEW.allocation_id);

        IF v_active_in_room >= v_capacity THEN
            RAISE EXCEPTION 'Trigger prevented SQL change: Room % is at full capacity (% beds).', NEW.room_id, v_capacity;
        END IF;
    END IF;

    RETURN NEW;
END;
$$;

CREATE OR REPLACE PROCEDURE pr_refresh_occupancy()
LANGUAGE plpgsql
AS $$
BEGIN
    INSERT INTO occupancy_snapshot (snapshot_date, room_id, occupied_beds, free_beds)
    SELECT
        CURRENT_DATE,
        r.room_id,
        COUNT(a.allocation_id)::INT                AS occupied_beds,
        (r.capacity - COUNT(a.allocation_id))::INT AS free_beds
    FROM rooms r
    LEFT JOIN allocations a
      ON a.room_id = r.room_id
     AND a.status = 'ACTIVE'
    GROUP BY r.room_id, r.capacity
    ON CONFLICT (snapshot_date, room_id)
    DO UPDATE SET
        occupied_beds = EXCLUDED.occupied_beds,
        free_beds     = EXCLUDED.free_beds;
END;
$$;
