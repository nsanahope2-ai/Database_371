CREATE OR REPLACE FUNCTION fn_loan_fine(
    p_loan_id INT,
    p_as_of   DATE
)
RETURNS NUMERIC(10, 2)
LANGUAGE plpgsql
AS $$
DECLARE
    v_due_on         DATE;
    v_returned_on    DATE;
    v_effective_date DATE;
    v_overdue_days   INTEGER;
BEGIN
    IF p_loan_id IS NULL THEN
        RAISE EXCEPTION 'Loan ID (p_loan_id) cannot be NULL.';
    END IF;

    IF p_as_of IS NULL THEN
        RAISE EXCEPTION 'As-of date (p_as_of) cannot be NULL.';
    END IF;

    SELECT due_on, returned_on
      INTO v_due_on, v_returned_on
      FROM loans
     WHERE loan_id = p_loan_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Loan ID % does not exist.', p_loan_id;
    END IF;

    v_effective_date := CASE
        WHEN v_returned_on IS NOT NULL THEN LEAST(p_as_of, v_returned_on)
        ELSE p_as_of
    END;

    v_overdue_days := GREATEST(v_effective_date - v_due_on, 0);

    RETURN (v_overdue_days * 2.00)::NUMERIC(10, 2);
END;
$$;

CREATE OR REPLACE PROCEDURE pr_return_loan(
    p_loan_id     INT,
    p_returned_on DATE
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_borrowed_on DATE;
    v_returned_on DATE;
    v_final_fine  NUMERIC(10, 2);
BEGIN
    IF p_loan_id IS NULL THEN
        RAISE EXCEPTION 'Loan ID (p_loan_id) cannot be NULL.';
    END IF;

    IF p_returned_on IS NULL THEN
        RAISE EXCEPTION 'Return date (p_returned_on) cannot be NULL for loan ID %.', p_loan_id;
    END IF;

    -- Lock target loan row to prevent concurrent double-return races
    SELECT borrowed_on, returned_on
      INTO v_borrowed_on, v_returned_on
      FROM loans
     WHERE loan_id = p_loan_id
       FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Loan ID % does not exist.', p_loan_id;
    END IF;

    IF v_returned_on IS NOT NULL THEN
        RAISE EXCEPTION 'Loan ID % has already been returned on %.', p_loan_id, v_returned_on;
    END IF;

    IF p_returned_on < v_borrowed_on THEN
        RAISE EXCEPTION 'Return date (%) cannot be earlier than borrowed_on (%) for loan ID %.',
            p_returned_on, v_borrowed_on, p_loan_id;
    END IF;

    -- Compute final fine as of p_returned_on before updating the row
    v_final_fine := fn_loan_fine(p_loan_id, p_returned_on);

    -- Persist returned_on and final fine together in a single atomic UPDATE
    UPDATE loans
       SET returned_on = p_returned_on,
           fine_amount = v_final_fine
     WHERE loan_id = p_loan_id;
END;
$$;

CREATE OR REPLACE PROCEDURE pr_refresh_overdue(p_as_of DATE)
LANGUAGE plpgsql
AS $$
DECLARE
    v_overdue_count INTEGER;
    v_total_fine    NUMERIC(12, 2);
BEGIN
    IF p_as_of IS NULL THEN
        RAISE EXCEPTION 'Snapshot date (p_as_of) cannot be NULL.';
    END IF;

    SELECT COUNT(*)::INTEGER,
           COALESCE(SUM(fn_loan_fine(l.loan_id, p_as_of)), 0.00)::NUMERIC(12, 2)
      INTO v_overdue_count, v_total_fine
      FROM loans l
     WHERE l.borrowed_on <= p_as_of
       AND l.due_on < p_as_of
       AND (l.returned_on IS NULL OR l.returned_on > p_as_of);

    INSERT INTO overdue_summary (snapshot_date, overdue_count, total_fine)
    VALUES (p_as_of, v_overdue_count, v_total_fine)
    ON CONFLICT (snapshot_date) DO UPDATE
        SET overdue_count = EXCLUDED.overdue_count,
            total_fine    = EXCLUDED.total_fine;
END;
$$;
